cmake_minimum_required(VERSION 3.20)
project(bmssp_native_foundation VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
cmake_path(IS_PREFIX CMAKE_SOURCE_DIR "${CMAKE_BINARY_DIR}" NORMALIZE BMSSP_BUILD_INSIDE_SOURCE)
if(BMSSP_BUILD_INSIDE_SOURCE)
  message(FATAL_ERROR "Use a build directory outside the source tree")
endif()
include(CTest)
option(BMSSP_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
add_library(bmssp_options INTERFACE)
target_compile_features(bmssp_options INTERFACE cxx_std_20)
if(MSVC)
  target_compile_options(bmssp_options INTERFACE /W4 /WX /permissive-)
else()
  target_compile_options(bmssp_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror)
endif()
if(BMSSP_SANITIZERS)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU" AND NOT WIN32)
    target_compile_options(bmssp_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(bmssp_options INTERFACE -fsanitize=address,undefined)
  else()
    message(FATAL_ERROR "Requested sanitizers require supported Clang/GNU on a non-Windows platform")
  endif()
endif()
add_library(bmssp_foundation native/src/graph.cpp native/src/reference_dijkstra.cpp native/src/path_reconstruction.cpp)
target_include_directories(bmssp_foundation PUBLIC native/include)
target_link_libraries(bmssp_foundation PUBLIC bmssp_options)
set_target_properties(bmssp_foundation PROPERTIES CXX_EXTENSIONS OFF)
add_executable(shortest_path_cli native/tools/shortest_path_cli.cpp)
target_link_libraries(shortest_path_cli PRIVATE bmssp_foundation)
if(BUILD_TESTING)
  foreach(category graph shortest_path path oracle)
    add_executable(${category}_tests native/tests/${category}_tests.cpp)
    target_link_libraries(${category}_tests PRIVATE bmssp_foundation)
    add_test(NAME ${category} COMMAND ${category}_tests)
  endforeach()
  add_test(NAME cli_smoke COMMAND shortest_path_cli --demo)
  set_tests_properties(graph shortest_path path oracle cli_smoke PROPERTIES TIMEOUT 60)
  set_tests_properties(cli_smoke PROPERTIES PASS_REGULAR_EXPRESSION "algorithm=reference_dijkstra reachable=true distance=7")
endif()
