cmake_minimum_required(VERSION 3.20)
project(bmssp_native_foundation VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
cmake_path(IS_PREFIX CMAKE_SOURCE_DIR "${CMAKE_BINARY_DIR}" NORMALIZE BMSSP_BUILD_INSIDE_SOURCE)
if(BMSSP_BUILD_INSIDE_SOURCE)
  message(FATAL_ERROR "Use a build directory outside the source tree")
endif()
include(CTest)
find_package(Threads REQUIRED)
option(BMSSP_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
add_library(bmssp_options INTERFACE)
target_compile_features(bmssp_options INTERFACE cxx_std_20)
if(MSVC)
  target_compile_options(bmssp_options INTERFACE /W4 /WX /permissive-)
else()
  target_compile_options(bmssp_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror)
endif()
if(BMSSP_SANITIZERS)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU" AND NOT WIN32)
    target_compile_options(bmssp_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(bmssp_options INTERFACE -fsanitize=address,undefined)
  else()
    message(FATAL_ERROR "Requested sanitizers require supported Clang/GNU on a non-Windows platform")
  endif()
endif()
add_library(bmssp_foundation native/src/graph.cpp native/src/reference_dijkstra.cpp native/src/path_reconstruction.cpp
  native/src/symbolic_label.cpp native/src/degree_reduction.cpp native/src/bounded_frontier.cpp
  native/src/bmssp_bounded.cpp native/src/bmssp_2025.cpp)
target_include_directories(bmssp_foundation PUBLIC native/include)
target_link_libraries(bmssp_foundation PUBLIC bmssp_options Threads::Threads)
set_target_properties(bmssp_foundation PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
add_executable(shortest_path_cli native/tools/shortest_path_cli.cpp)
target_link_libraries(shortest_path_cli PRIVATE bmssp_foundation)
add_executable(bmssp_benchmark native/tools/bmssp_benchmark.cpp)
target_link_libraries(bmssp_benchmark PRIVATE bmssp_foundation)
if(BUILD_TESTING)
  foreach(category graph shortest_path path oracle degree_reduction symbolic_label frontier bounded bmssp numerical)
    add_executable(${category}_tests native/tests/${category}_tests.cpp)
    target_link_libraries(${category}_tests PRIVATE bmssp_foundation)
    add_test(NAME ${category} COMMAND ${category}_tests)
  endforeach()
  if(UNIX)
    add_executable(arena_memory_tests native/tests/arena_memory_tests.cpp)
    target_link_libraries(arena_memory_tests PRIVATE bmssp_foundation)
    add_test(NAME arena_memory COMMAND arena_memory_tests)
    set_tests_properties(arena_memory PROPERTIES TIMEOUT 120)
  endif()
  add_test(NAME cli_smoke COMMAND shortest_path_cli --demo)
  add_test(NAME bmssp_cli_smoke COMMAND shortest_path_cli --bmssp-demo)
  set_tests_properties(graph shortest_path path oracle cli_smoke PROPERTIES TIMEOUT 60)
  set_tests_properties(degree_reduction symbolic_label frontier bounded bmssp numerical bmssp_cli_smoke PROPERTIES TIMEOUT 120)
  set_tests_properties(bmssp_cli_smoke PROPERTIES PASS_REGULAR_EXPRESSION "algorithm=bmssp_2025 reachable=true distance=7")
  set_tests_properties(cli_smoke PROPERTIES PASS_REGULAR_EXPRESSION "algorithm=reference_dijkstra reachable=true distance=7")
endif()

# Optional: native-only builds retain zero Python dependencies.
option(BMSSP_PYTHON "Build the independent bmssp_native Python extension" OFF)
if(BMSSP_PYTHON)
  set(PYBIND11_FINDPYTHON ON)
  find_package(Python 3.9 REQUIRED COMPONENTS Interpreter Development.Module)
  find_package(pybind11 2.11 CONFIG REQUIRED)
  pybind11_add_module(bmssp_native NO_EXTRAS native/python/bindings.cpp)
  target_link_libraries(bmssp_native PRIVATE bmssp_foundation)
  set_target_properties(bmssp_native PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/python")
  if(BUILD_TESTING AND NOT BMSSP_SANITIZERS)
    add_test(NAME python_binding_smoke COMMAND "${Python_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/native/python/import_smoke.py")
    set_tests_properties(python_binding_smoke PROPERTIES ENVIRONMENT "PYTHONPATH=$<TARGET_FILE_DIR:bmssp_native>" TIMEOUT 60)
  endif()
endif()

# Allocation interception is isolated to this explicit probe executable, never
# linked into production libraries, bindings, or latency benchmark processes.
option(BMSSP_ALLOCATION_PROBE "Build ordinary-C++ allocation probe (Unix)" OFF)
if(BMSSP_ALLOCATION_PROBE)
  if(NOT UNIX)
    message(FATAL_ERROR "The allocation probe currently requires Unix getrusage")
  endif()
  add_executable(allocation_probe native/tools/allocation_probe.cpp)
  target_link_libraries(allocation_probe PRIVATE bmssp_foundation)
endif()
